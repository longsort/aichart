f_kind() =>
    string per = timeframe.period
    per == "W" or per == "M" ? "WM" : per == "D" ? "D" : per == "240" ? "4H" : per == "60" ? "1H" : per == "15" ? "15" : per == "5" ? "5" : per == "3" ? "3" : per == "1" ? "1m" : "LTF"

f_ko() =>
    string k = f_kind()
    k == "WM" ? (timeframe.period == "M" ? "1월" : "1주") : k == "D" ? "1일" : k == "4H" ? "4시간" : k == "1H" ? "1시간" : k == "15" ? "15분" : k == "5" ? "5분" : k == "3" ? "3분" : k == "1m" ? "1분" : timeframe.period

f_look() =>
    int n = bar_index + 1
    string k = f_kind()
    int cap = k == "WM" ? 24 : k == "D" ? 32 : k == "4H" ? 40 : k == "1H" ? 48 : 36
    math.max(4, math.min(math.max(1, n - 2), cap))

f_atr() =>
    float tr = math.max(high - low, math.abs(high - nz(close[1], close)), math.abs(low - nz(close[1], close)))
    float s = ta.sma(tr, 14)
    s > 0 ? s : close * 0.006

f_band(float mid, float atr) =>
    string k = f_kind()
    float mult = k == "WM" ? 0.06 : k == "D" ? 0.07 : k == "4H" ? 0.09 : k == "1H" ? 0.10 : k == "15" ? 0.11 : k == "1m" or k == "3" ? 0.14 : 0.12
    float cap = k == "WM" ? 0.0032 : k == "D" ? 0.003 : k == "4H" ? 0.0036 : k == "1H" ? 0.004 : k == "15" ? 0.0045 : k == "1m" or k == "3" ? 0.006 : 0.0055
    float raw = math.max(atr * mult, math.abs(mid) * 0.0005, 18.0)
    math.min(raw, math.abs(mid) * cap)

f_scan(int look, float last) =>
    float deep = na
    float below1 = na
    float below2 = na
    for i = 1 to look
        float b = low[i]
        float a = low[i + 1]
        float c = nz(low[i - 1], b)
        if b > 0 and b <= a and b <= c
            deep := na(deep) ? b : math.min(deep, b)
            if b < last * 0.998
                if na(below1) or b < below1
                    below2 := below1
                    below1 := b
                else if na(below2) or b < below2
                    below2 := b
    float slo = ta.lowest(low[1], math.max(2, look - 1))
    if slo > 0
        deep := na(deep) ? slo : math.min(deep, slo)
        if slo < last * 0.998
            if na(below1) or slo < below1
                below2 := below1
                below1 := slo
            else if na(below2) or slo < below2
                below2 := slo
    [deep, below1, below2]

f_floor() =>
    string k = f_kind()
    int n = bar_index + 1
    int need = k == "WM" or k == "D" ? 8 : k == "4H" or k == "1H" ? 12 : 16
    float lo = na
    float hi = na
    float atr = f_atr()
    int look = f_look()
    float last = close
    int swingLen = math.max(2, look - 1)
    float swingHi = ta.highest(high[1], swingLen)
    [deep, below1, below2] = f_scan(look, last)
    float rangeLo = ta.lowest(low[1], swingLen)
    if na(deep)
        deep := rangeLo
    if n >= need and close > 0
        float prefer = na
        if k == "WM"
            prefer := deep > 0 and deep < last * 0.9995 ? deep : na
        else if k == "D"
            float d2 = na
            for i = 1 to look
                float b = low[i]
                float a = low[i + 1]
                float c = nz(low[i - 1], b)
                if b > 0 and b <= a and b <= c and b < last * 0.997 and deep > 0 and b > deep * 1.01
                    d2 := na(d2) ? b : math.min(d2, b)
            prefer := d2
            if na(prefer) and deep > 0 and deep < last - atr * 1.5
                prefer := math.min(last - atr * 1.15, deep + math.max(atr * 1.6, last * 0.014))
        else if k == "4H"
            prefer := below2
        else
            prefer := below1
        if na(prefer) or prefer <= 0 or prefer >= last * 0.9995
            float kk = k == "D" ? 1.35 : k == "4H" ? 1.0 : k == "WM" ? 0.7 : 0.55
            prefer := last - math.max(atr * kk, last * 0.008)
        if prefer > last - math.max(atr * 0.25, last * 0.0025)
            prefer := last - math.max(atr * 0.55, last * 0.006)
        bool distant = prefer < last - math.max(atr * 0.35, last * 0.004)
        bool allow = k == "D" or k == "4H" or k == "WM" or k == "1m"
        bool ok = true
        if not distant
            float dropMin = k == "WM" or k == "D" or k == "4H" ? 0.01 : k == "1H" ? 0.015 : 0.02
            float drop = swingHi > 0 ? (swingHi - last) / swingHi : 0.0
            if drop < dropMin
                ok := allow
            float resist = prefer + atr * 0.65
            if ok and last > resist + atr * 0.2
                ok := allow
        if ok and prefer > 0
            float half = f_band(prefer, atr)
            lo := prefer - half / 2.0
            hi := prefer + half / 2.0
    [lo, hi]

f_ceil() =>
    string k = f_kind()
    int n = bar_index + 1
    int need = k == "WM" or k == "D" ? 8 : k == "4H" or k == "1H" ? 12 : 16
    float lo = na
    float hi = na
    float atr = f_atr()
    float last = close
    int look = k == "WM" ? math.min(math.max(n - 2, 2), 24) : math.min(math.max(n - 2, 2), 60)
    float swingHi = ta.highest(high[1], math.max(2, look))
    if n >= need and close > 0
        float ceil = swingHi > last * 1.001 ? swingHi : last + atr * 0.55
        bool allow = k == "D" or k == "4H" or k == "WM" or k == "1H" or k == "15" or k == "1m"
        if ceil <= last and allow
            ceil := last + math.max(atr * 0.55, last * 0.008)
        float maxDist = math.max(atr * (k == "WM" ? 8.0 : k == "D" ? 6.0 : 4.5), last * 0.08)
        if ceil > last + maxDist
            ceil := last + maxDist * 0.85
        if ceil <= last * 1.0003 and allow
            ceil := swingHi > last * 1.0005 ? swingHi : last + math.max(atr * 0.55, last * 0.008)
        if ceil > last

f_box(box face, bool on, float lo, float hi, color bd, string txt, color ink) =>
    box f = face
    if not on or na(lo) or na(hi) or hi <= lo
        if not na(f)
            box.delete(f)
        f := na
    else
        int xL = bar_index - boxLeft
        int xR = bar_index + boxRight
        if na(f)
            f := box.new(xL, hi, xR, lo, border_color = bd, bgcolor = color.new(bd, zoneOpac), border_width = zoneW, text = txt, text_color = ink, text_halign = text.align_left, text_valign = text.align_center, text_size = size.tiny, force_overlay = true)
        else
            box.set_left(f, xL)
            box.set_right(f, xR)
            box.set_top(f, hi)
            box.set_bottom(f, lo)
            box.set_border_color(f, bd)
            box.set_bgcolor(f, color.new(bd, zoneOpac))
            box.set_border_width(f, zoneW)
            box.set_text(f, txt)
            box.set_text_color(f, ink)
            box.set_text_valign(f, text.align_center)

            box.set_text_halign(f, text.align_center)

            box.set_text_halign(f, text.align_left)

            box.set_text_size(f, size.normal)
        else if txtSz == "보통"
            box.set_text_size(f, size.small)
        else
            box.set_text_size(f, size.tiny)
    f

f_sw() =>
    float ph = ta.pivothigh(high, swLeft, swRight)
    float pl = ta.pivotlow(low, swLeft, swRight)
    var float hiP = na
    var float loP = na
    if not na(ph)
        hiP := ph
    if not na(pl)
        loP := pl
    float a = ta.atr(14)
    float vs = ta.sma(volume, 20)
    float rv = vs > 0 ? volume / vs : 1.0
    float pierce = math.max(nz(a, close * 0.006) * swAtr, close * 0.0004)
    bool bear = not na(hiP) and high > hiP + pierce and close < hiP and close < open and nz(close[1], close) <= hiP and rv >= 0.9
    bool bull = not na(loP) and low < loP - pierce and close > loP and close > open and nz(close[1], close) >= loP and rv >= 0.9
    int dir = bear and not bull ? -1 : bull and not bear ? 1 : 0
    float y = dir < 0 ? high : dir > 0 ? low : na
    var int mem = 0
    var float memY = na
    var int memT = na
    if dir != 0
        mem := dir
        memY := y
        memT := time
    int age = ta.barssince(dir != 0)
    [mem, memY, age, memT]

f_barAt(int tms, float y, int dir) =>
    int hit = bar_index
    if not na(tms) and not na(y) and dir != 0
        float best = 1000000000.0
        int n = math.min(bar_index, 480)
        for i = 0 to n
            if time[i] < tms
                break
            float py = dir > 0 ? low[i] : high[i]
            float dist = math.abs(py - y)
            if dist < best
                best := dist
                hit := bar_index - i
    hit

f_volTxt(float rv, float bp, float bodyR, float chg, float slope) =>
    float sp = 1.0 - bp
    string t = "약함"
    if rv >= 2.35 and bodyR < 0.4
        t := "흡수"
    else if rv >= 2.55 and bodyR >= 0.55
        t := chg >= 0 ? "매수절정" : "매도절정"
    else if rv >= 1.55 and bodyR >= 0.52 and math.abs(chg) >= 0.08
        t := chg >= 0 ? "돌파량" : "이탈량"
    else if rv < 0.68 and chg > 0.02
        t := "수요없음"
    else if rv < 0.68 and chg < -0.02
        t := "공급없음"
    else if slope > 0.06 and sp >= 0.56
        t := "수급괴리"
    else if slope < -0.06 and bp >= 0.56
        t := "수급괴리"
    else if slope > 0.05 and bp >= 0.56 and rv >= 1.05
        t := "수급동의↑"
    else if slope < -0.05 and sp >= 0.56 and rv >= 1.05
        t := "수급동의↓"
    else if bp >= 0.56
        t := "매수우세"
    else if sp >= 0.56
        t := "매도우세"
    int bn = math.round(bp * 100)
    int sn = math.round(sp * 100)
    string mv = math.abs(chg) >= 0.35 ? (chg >= 0 ? "↑" : "↓") + str.tostring(math.abs(chg), "#.#") : ""
    string flow = sp >= 0.56 ? "매도" + str.tostring(sn) + mv : "매" + str.tostring(bn) + mv
    string txt = t == "약함" ? flow : t + " " + flow
    bool hot = (t != "약함" and t != "매수우세" and t != "매도우세") or rv >= 1.65
    hot ? txt : ""

    if ok

        float o1 = useLow ? low[ago + 1] : high[ago + 1]
        float o2 = useLow ? low[ago + 2] : high[ago + 2]

    ok

    if n >= 14
        for k = 1 to 14

    if n >= 6 and atrNow > 0
        for i = n to 2
            float c3h = high[i]
            float c3l = low[i]
            float c1h = high[i + 2]
            float c1l = low[i + 2]
            int jStop = math.max(1, i - 36)
            if c1h < c3l
                float glo = c1h
                float ghi = c3l
                float gh = ghi - glo
                if gh >= syminfo.mintick * 2 and gh <= math.max(atrNow * 3.0, close * 0.02)
                    bool mit = false
                    if i > 1
                        for j = i - 1 to jStop
                            if low[j] <= ghi and high[j] >= glo
                                mit := true
                                break
                    if not mit
                        float dist = math.abs((glo + ghi) / 2.0 - close)
                        if dist < bullD or (dist == bullD and i < bullAgo)
                            bullD := dist
                            bullAgo := i
                            bullLo := glo
                            bullHi := ghi
                    if i <= 36 and array.size(upLo) < 8
                        array.push(upLo, glo)
                        array.push(upHi, ghi)

f_gapPick(bool on, float lo, float hi, bool wantBelow) =>
    float outLo = na
    float outHi = na
    float best = 1000000000.0
    bool ok = on and not na(lo) and not na(hi) and hi > lo
    if ok
        float mid = (lo + hi) / 2.0
        bool side = wantBelow ? mid <= close : mid > close
        if side
            best := math.abs(mid - close)
            outLo := lo
            outHi := hi
    [outLo, outHi, best]

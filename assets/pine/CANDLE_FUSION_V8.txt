// 01 — 20 candlestick patterns combined with normalized candle anatomy.
float atr=ta.atr(14)
float rangeC=math.max(high-low,syminfo.mintick)
float bodyC=math.abs(close-open)
float bodyQ=bodyC/rangeC
float upWick=(high-math.max(open,close))/rangeC
float dnWick=(math.min(open,close)-low)/rangeC
float closeQ=(close-low)/rangeC
bool bull=close>open
bool bear=close<open
bool strongBull=bull and bodyQ>=minBody and closeQ>=0.72
bool strongBear=bear and bodyQ>=minBody and closeQ<=0.28
float prevBody=math.abs(close[1]-open[1])
float prevQ=prevBody/math.max(high[1]-low[1],syminfo.mintick)
bool doji=bodyQ<=0.12
bool engulfL=bull and close[1]<open[1] and open<=close[1] and close>=open[1]
bool engulfS=bear and close[1]>open[1] and open>=close[1] and close<=open[1]
bool hammer=dnWick>=minWick and dnWick>=2.0*bodyQ and upWick<=0.20

// 02 — VOLUME vs candle progress: pressure, exhaustion and absorption.
float volumeMA=ta.sma(volume,20)
float rvol=volumeMA>0?volume/volumeMA:na
float upFrac=f_clip(closeQ,0.0,1.0)
float buyProxy=nz(volume,0)*upFrac
float sellProxy=nz(volume,0)*(1-upFrac)
float deltaProxy=buyProxy-sellProxy // rough candle-position proxy, NOT true CVD
float cvdProxy=math.sum(deltaProxy,34)
float velocity=(close-nz(close[1],close))/math.max(atr,syminfo.mintick)
float pressure=f_clip(0.42*(2*closeQ-1)+0.35*(bull?bodyQ:bear?-bodyQ:0)+0.23*(dnWick-upWick),-1,1)
float effort=math.max(nz(rvol,1),0.1)
float efficiency=math.abs(velocity)/effort
bool absorbL=not na(rvol) and rvol>=1.35 and (dnWick>=0.31 or closeQ>0.68) and efficiency<0.38 and closeQ>0.55
bool absorbS=not na(rvol) and rvol>=1.35 and (upWick>=0.31 or closeQ<0.32) and efficiency<0.38 and closeQ<0.45
bool progressFailureL=not na(rvol) and rvol>1.35 and low<low[1] and close>low[1] and closeQ>0.60
bool progressFailureS=not na(rvol) and rvol>1.35 and high>high[1] and close<high[1] and closeQ<0.40
bool volumeL=not na(rvol) and rvol>=minVol and (pressure>0.12 or absorbL or progressFailureL)
bool volumeS=not na(rvol) and rvol>=minVol and (pressure< -0.12 or absorbS or progressFailureS)

// 07B — Event-origin persistent OB/FVG/iFVG zones (NOT last-bar decorations).
// New zones originate only on a CONFIRMED event bar. The original price and
// left edge never migrate. Touch changes border; invalidation freezes right edge.
type FusionZone
    box draw
    float top
    float bot
    int side
    int src
    int born
    int state
    int touches
    int lastTouchAt
    float quality
    int bornTime
    int lastStudyAt
var array<FusionZone> fusionZones=array.new<FusionZone>()
f_zoneAdd(array<FusionZone> store,float top,float bot,int direction,int source,float strength,int beginX,int objectLimit)=>
    if not na(top) and not na(bot) and top>bot+syminfo.mintick
        bool exists=false
        if array.size(store)>0
            for i=0 to array.size(store)-1
                FusionZone old=array.get(store,i)
                if old.state==0 and old.src==source and old.side==direction and math.abs(old.top-top)<=syminfo.mintick*2 and math.abs(old.bot-bot)<=syminfo.mintick*2
                    exists:=true
                    break
        if not exists
            color zc=direction==1?color.rgb(22,199,116):color.rgb(242,70,87)
            box b=box.new(left=beginX,top=top,right=bar_index+2,bottom=bot,xloc=xloc.bar_index,
                 border_color=color.new(zc,100),
                 bgcolor=color.new(zc,100),border_width=1)
            array.push(store,FusionZone.new(b,top,bot,direction,source,bar_index,0,0,bar_index,strength,time,-100000))
        while array.size(store)>objectLimit
            // Preserve active structural zones before obsolete history.
            int dropId=0
            if array.size(store)>0
                for q=0 to array.size(store)-1
                    FusionZone candidate=array.get(store,q)
                    if candidate.state!=0
                        dropId:=q
                        break
            FusionZone rem=array.get(store,dropId)
            box.delete(rem.draw)
            array.remove(store,dropId)
    0

f_v7Wilson(int successes,int n) =>
    float lo=na
    float hi=na
    if n>0
        float z=1.96
        float p=float(successes)/n
        float d=1.0+z*z/n
        float mid=(p+z*z/(2.0*n))/d
        float delta=z*math.sqrt((p*(1.0-p)+z*z/(4.0*n))/n)/d
        lo:=100.0*math.max(0.0,mid-delta)
        hi:=100.0*math.min(1.0,mid+delta)
    [lo,hi]

bool lineReadyS=not na(swingHigh) and not na(oldSwingHigh) and not na(swingHighAt) and not na(oldHighAt) and swingHighAt>oldHighAt

// 08 — Confirmed HTF pack: true pivots / OB / FVG / composite blue reference.
// No same- or lower-TF request ever treated as a confirmed HTF signal.
f_htf() =>

// Price-cluster algorithm: choose a location CLOSE to the candle's contact
// wick, not the high-score but distant level. Independent family types count
// only once, and a 'structural' location is mandatory for an entry candidate.
f_cluster(int side,array<float> levels,array<int> dirs,array<int> families,array<float> weights,array<string> names,float candleEdge,float currentClose,float a,float tol)=>
    float anchor=na
    float priority=-1.0
    string desc=""
    if array.size(levels)>0
        for j=0 to array.size(levels)-1
            float p=array.get(levels,j)
            int d=array.get(dirs,j)
            int actual=d==0?(p<=currentClose?1:-1):d
            float distance=math.abs(p-candleEdge)
            if actual==side and distance<=a*1.15 and array.get(families,j)<=4
                float prox=f_clip(1-distance/(a*1.15),0,1)
                float rank=0.55*array.get(weights,j)+0.45*prox
                if rank>priority
                    priority:=rank
                    anchor:=p
                    desc:=array.get(names,j)
    bool f1=false
    bool f2=false
    bool f3=false
    bool f4=false
    bool f5=false
    bool f6=false
    bool f7=false
    bool f8=false
    float strength=0.0
    if not na(anchor)
        for j=0 to array.size(levels)-1
            float p=array.get(levels,j)
            int d=array.get(dirs,j)
            int actual=d==0?(p<=currentClose?1:-1):d
            if actual==side and math.abs(p-anchor)<=a*tol
                int f=array.get(families,j)
                f1:=f1 or f==1
                f2:=f2 or f==2
                f3:=f3 or f==3
                f4:=f4 or f==4
                f5:=f5 or f==5
                f6:=f6 or f==6
                f7:=f7 or f==7
                f8:=f8 or f==8
                strength:=math.max(strength,array.get(weights,j))
    int count=(f1?1:0)+(f2?1:0)+(f3?1:0)+(f4?1:0)+(f5?1:0)+(f6?1:0)+(f7?1:0)+(f8?1:0)
    bool structural=f1 or f2 or f3 or f4 or f7
    bool realOrderBlock=f3 or f4
    [anchor,count,structural,realOrderBlock,strength,desc]

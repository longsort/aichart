    string picked = "12M"
    array<string> ladder = array.from("60", "240", "1D", "1W", "1M", "3M", "12M")

        if not na(s) and s > minSecExclusive
            picked := t
            break

        eq_bullFvgLo := high[2]
        eq_bullFvgHi := low
    if high < low[2] and low[2]-high >= eq_atr*0.18
        eq_bearFvgLo := high
        eq_bearFvgHi := low[2]
    if not na(eq_bullFvgLo) and close < eq_bullFvgLo
        eq_invRes := (eq_bullFvgLo+eq_bullFvgHi)*0.5
        eq_bullFvgLo := na
        eq_bullFvgHi := na
    if not na(eq_bearFvgHi) and close > eq_bearFvgHi
        eq_invSup := (eq_bearFvgLo+eq_bearFvgHi)*0.5
        eq_bearFvgLo := na
        eq_bearFvgHi := na
    if not na(eq_bullFvgLo) and not na(eq_bearFvgLo)
        float oLo = math.max(eq_bullFvgLo,eq_bearFvgLo)
        float oHi = math.min(eq_bullFvgHi,eq_bearFvgHi)
        eq_bprMid := oHi>oLo ? (oLo+oHi)*0.5 : na

// Confirmed swing anchored ratios; 1x1 is normalized price-range / bar-range,
// not a geometric 45-degree screen angle.
float eq_swingRange = not na(eq_pH1) and not na(eq_pL1) ? math.abs(eq_pH1-eq_pL1) : na
int eq_swingBars = not na(eq_xH1) and not na(eq_xL1) ? math.max(1,math.abs(eq_xH1-eq_xL1)) : 0
float eq_gannStep = eq_swingBars>0 ? eq_swingRange/eq_swingBars : na
float eq_gannFanL = eq_useGann and not na(eq_gannStep) and not na(eq_pL1) ? eq_pL1+(bar_index-eq_xL1)*eq_gannStep : na
float eq_gannFanS = eq_useGann and not na(eq_gannStep) and not na(eq_pH1) ? eq_pH1-(bar_index-eq_xH1)*eq_gannStep : na
float eq_gannOne4 = not na(eq_swingRange) ? eq_pL1+eq_swingRange*0.25 : na
float eq_gannThree4 = not na(eq_swingRange) ? eq_pL1+eq_swingRange*0.75 : na
float eq_gannHalf = not na(eq_swingRange) ? eq_pL1+eq_swingRange*0.50 : na
int eq_sinceTurn = array.size(eq_turnX)>0 ? bar_index-array.get(eq_turnX,array.size(eq_turnX)-1) : 0
bool eq_cycleHit = eq_useTime and eq_swingBars>=6 and eq_sinceTurn>0 and (eq_sinceTurn%eq_swingBars<=1 or eq_sinceTurn%eq_swingBars>=eq_swingBars-1)

// Harmonic: geometrical XABCD-like 5-pivot and AB=CD completion candidates.
// Label only at CONFIRMATION of D; no claim of exact Gartley/Bat/Cypher ratios.
float eq_harmLow = na
float eq_harmHigh = na
bool eq_ellUp = false
bool eq_ellDn = false
int eq_nTurn = array.size(eq_turns)
if eq_nTurn >= 4
    float a = array.get(eq_turns,eq_nTurn-4)
    float b = array.get(eq_turns,eq_nTurn-3)
    float c = array.get(eq_turns,eq_nTurn-2)
    float d = array.get(eq_turns,eq_nTurn-1)
    float ab = math.abs(b-a)
    float bc = math.abs(c-b)
    float cd = math.abs(d-c)
    float retr = ab>0 ? bc/ab : 0.0
    bool abcd = eq_useHarm and ab>0 and cd>0 and retr>=0.382 and retr<=0.886 and math.abs(cd/ab-1.0)<=eq_patternTol
    if abcd
        if array.get(eq_turnDir,eq_nTurn-1)==-1
            eq_harmLow := d
        else
            eq_harmHigh := d
if eq_nTurn >= 5
    float x = array.get(eq_turns,eq_nTurn-5)
    float a2 = array.get(eq_turns,eq_nTurn-4)
    float b2 = array.get(eq_turns,eq_nTurn-3)
    float c2 = array.get(eq_turns,eq_nTurn-2)
    float d2 = array.get(eq_turns,eq_nTurn-1)
    float xa = math.abs(a2-x)
    float ab2 = math.abs(b2-a2)
    float bc2 = math.abs(c2-b2)
    float cd2 = math.abs(d2-c2)
    bool ratioCandidate = eq_useHarm and xa>0 and ab2/xa>=0.38 and ab2/xa<=0.90 and bc2/math.max(ab2,syminfo.mintick)>=0.382 and bc2/math.max(ab2,syminfo.mintick)<=0.886 and cd2/math.max(bc2,syminfo.mintick)>=1.13 and cd2/math.max(bc2,syminfo.mintick)<=2.62
    if ratioCandidate
        if array.get(eq_turnDir,eq_nTurn-1)==-1
            eq_harmLow := d2
        else
            eq_harmHigh := d2
    // Five consecutive eq_confirmed eq_turns: basic directional wave structure.
    // Not a full Elliott count; use ONE separate family max.
    eq_ellUp := eq_useElliott and array.get(eq_turnDir,eq_nTurn-1)==-1 and d2>b2 and c2>a2
    eq_ellDn := eq_useElliott and array.get(eq_turnDir,eq_nTurn-1)==1 and d2<b2 and c2<a2

// Additional-family scoring uses each geometry class at most ONCE.
eq_f_hits(float px, bool isLong) =>
    int hits = 0
    if not na(px) and not na(eq_tol) and eq_tol>0
        bool geom = eq_useGann and (math.abs(px-nz(eq_gannFanL,1e20))<=eq_tol or math.abs(px-nz(eq_gannFanS,1e20))<=eq_tol or math.abs(px-nz(eq_gannOne4,1e20))<=eq_tol or math.abs(px-nz(eq_gannHalf,1e20))<=eq_tol or math.abs(px-nz(eq_gannThree4,1e20))<=eq_tol)
        bool pat = eq_usePattern and (eq_fallingWedge or eq_triangle or eq_risingWedge) and (math.abs(px-nz(eq_trendHi,1e20))<=eq_tol or math.abs(px-nz(eq_trendLo,1e20))<=eq_tol)
        bool h = eq_useHarm and math.abs(px-nz(isLong ? eq_harmLow : eq_harmHigh,1e20))<=eq_tol
        bool e = eq_useElliott and (isLong ? eq_ellUp : eq_ellDn) and math.abs(px-nz(isLong ? eq_pL1 : eq_pH1,1e20))<=eq_tol*2
        bool gapState = eq_useGapState and (math.abs(px-nz(isLong ? eq_invSup : eq_invRes,1e20))<=eq_tol or math.abs(px-nz(eq_bprMid,1e20))<=eq_tol)
        bool timeState = eq_cycleHit and (isLong ? close >= px-eq_tol : close <= px+eq_tol)
        hits := (geom?1:0)+(pat?1:0)+(h?1:0)+(e?1:0)+(gapState?1:0)+(timeState?1:0)
    hits

// Pick the candidate matching current price vicinity with the greatest UNIQUE
// advanced-family count. Base price anchors are NOT themselves new families.
float eq_support = na
float eq_resistance = na
int eq_extraLong = 0
int eq_extraShort = 0
array<float> eq_supCandidates = array.from(eq_pL1,eq_bullFvgLo,eq_invSup,low[1])

for i=0 to array.size(eq_supCandidates)-1

    if not na(px) and px<=close+eq_tol and (close-px)<=eq_atr*eq_rangeMaxATR
        int hc = eq_f_hits(px,true)
        if na(eq_support) or hc>eq_extraLong or hc==eq_extraLong and math.abs(close-px)<math.abs(close-eq_support)
            eq_support := px
            eq_extraLong := hc
for i=0 to array.size(eq_resCandidates)-1
    float px = array.get(eq_resCandidates,i)
    if not na(px) and px>=close-eq_tol and (px-close)<=eq_atr*eq_rangeMaxATR
        int hc = eq_f_hits(px,false)
        if na(eq_resistance) or hc>eq_extraShort or hc==eq_extraShort and math.abs(px-close)<math.abs(eq_resistance-close)
            eq_resistance := px
            eq_extraShort := hc

// --------------------------------------------------------------------------
// The level is frozen at the first confirmed break. A second candle must
// remain beyond it for acceptance. A subsequent violation marks failure.
// Do not use barstate.isconfirmed inside request.security(). Instead the
// wrapper below requests [1] + lookahead_on for STRICTLY higher TFs.
// State codes: 0 neutral, +1/-1 break pending, +2/-2 held/accepted,
// +3/-3 retest held, +4/-4 acceptance FAILURE, +5/-5 sweep/rejection,
// 98 insufficient data, 99 lower timeframe (change chart to assess).
f_mtfDetect() =>
    var int phase = 0
    var float level = na
    var int age = 0
    int state = 98
    float localAtr = ta.atr(14)
    float margin = localAtr * mtfCloseBufferATR
    float priorHi = ta.highest(high[1], mtfRangeBars)
    float priorLo = ta.lowest(low[1], mtfRangeBars)
    if not na(localAtr) and not na(priorHi) and not na(priorLo)
        state := 0
        if phase == 0
            level := na
            age := 0
            if high > priorHi + margin and close <= priorHi
                state := -5
                level := priorHi
            else if low < priorLo - margin and close >= priorLo
                state := 5
                level := priorLo
            else if close > priorHi + margin
                phase := 1
                level := priorHi
                state := 1
            else if close < priorLo - margin
                phase := -1
                level := priorLo
                state := -1
        else if phase == 1
            age += 1
            if close < level - margin
                state := 4
                phase := 0
            else if close > level + margin and age >= 1
                phase := 2
                state := low <= level + margin ? 3 : 2
            else
                state := 1
        else if phase == 2
            if close < level - margin
                state := 4
                phase := 0
            else
                state := low <= level + margin and close > level ? 3 : 2
        else if phase == -1
            age += 1
            if close > level + margin
                state := -4
                phase := 0
            else if close < level - margin and age >= 1
                phase := -2
                state := high >= level - margin ? -3 : -2
            else
                state := -1
        else if phase == -2
            if close > level + margin
                state := -4
                phase := 0
            else
                state := high >= level - margin and close < level ? -3 : -2
    [state, level]

// Expression wrapper: the three values refer to a PREVIOUS COMPLETED HTF bar.
f_mtfConfirmed() =>
    [stateAtTf, levelAtTf] = f_mtfDetect()
    [stateAtTf[1], levelAtTf[1], time_close[1]]

f_regResidual(float fitted, float slope, int n) =>
    float sumSq = 0.0
    for i = 0 to n - 1
        float fittedPast = fitted - slope * i
        float e = close[i] - fittedPast
        sumSq += e * e
    math.sqrt(sumSq / n)

// Proxy: 65% of each bar's volume assigned to HLC3 bin, 35% to close bin.
// Not executable-volume distribution. Keeps cost O(bars + bins) per candle.
f_profile(int bars, int binCount, float coverage) =>
    float resultPOC = na
    float resultVAH = na
    float resultVAL = na
    float highestP = ta.highest(high, bars)
    float lowestP = ta.lowest(low, bars)
    float stepP = (highestP - lowestP) / binCount
    if bar_index >= bars and stepP > syminfo.mintick * 0.01
        array<float> histogram = array.new_float(binCount, 0.0)
        float totalV = 0.0
        for i = 0 to bars - 1
            float v = nz(volume[i], 0.0)
            if v > 0 and not na(hlc3[i])
                int hBin = math.max(0, math.min(binCount - 1, int(math.floor((hlc3[i] - lowestP) / stepP))))
                int cBin = math.max(0, math.min(binCount - 1, int(math.floor((close[i] - lowestP) / stepP))))
                array.set(histogram, hBin, array.get(histogram, hBin) + v * 0.65)
                array.set(histogram, cBin, array.get(histogram, cBin) + v * 0.35)
                totalV += v
        if totalV > 0
            int pocBin = 0
            float maxV = -1.0
            for i = 0 to binCount - 1
                if array.get(histogram, i) > maxV
                    maxV := array.get(histogram, i)
                    pocBin := i
            int left = pocBin
            int right = pocBin
            float covered = array.get(histogram, pocBin)
            while covered < totalV * coverage and (left > 0 or right < binCount - 1)
                float leftV = left > 0 ? array.get(histogram, left - 1) : -1.0
                float rightV = right < binCount - 1 ? array.get(histogram, right + 1) : -1.0
                if leftV > rightV
                    left -= 1
                    covered += leftV
                else
                    right += 1
                    covered += rightV
            resultPOC := lowestP + (pocBin + 0.5) * stepP
            resultVAL := lowestP + left * stepP
            resultVAH := lowestP + (right + 1) * stepP
    [resultPOC, resultVAH, resultVAL]

// A merge requires actual price-range intersection, similar widths,
// same side, non-invalid state AND proximity. Original prices are frozen.
f_putZone(array<float> tops, array<float> bottoms, array<int> sides,
          array<int> borns, array<int> sources, array<int> strength,
          array<int> breaks, array<int> flipStates, array<int> flipAt,
          array<int> uidArr, array<int> roleVers, array<int> gradeSince,
          float top, float bot, int side, int src, float tolerance, int maximum) =>
    bool inserted = false
    if not na(top) and not na(bot) and top > bot and bot > 0
        int matchId = -1
        if array.size(tops) > 0
            for k = 0 to array.size(tops) - 1
                float oldTop = array.get(tops,k)
                float oldBot = array.get(bottoms,k)
                float overlap = math.min(oldTop, top) - math.max(oldBot, bot)
                float minWidth = math.min(oldTop-oldBot, top-bot)
                float maxWidth = math.max(oldTop-oldBot, top-bot)
                float oldMid = (oldTop+oldBot)*0.5
                float newMid = (top+bot)*0.5
                bool compatible = array.get(sides,k) == side and array.get(flipStates,k) == 0
                bool shapeOK = minWidth > 0 and overlap >= minWidth*0.25 and maxWidth <= minWidth*3.0
                if compatible and shapeOK and math.abs(oldMid-newMid) <= tolerance
                    matchId := k
                    break
        if matchId >= 0
            int oldMask = array.get(sources,matchId)
            if oldMask % src != 0
                int newMask = oldMask*src
                array.set(sources,matchId,newMask)
                array.set(strength,matchId,math.max(1,f_sourceCount(newMask)))
                array.set(gradeSince,matchId,bar_index)
        else
            if array.size(tops) >= maximum
                array.shift(tops)
                array.shift(bottoms)
                array.shift(sides)
                array.shift(borns)
                array.shift(sources)
                array.shift(strength)
                array.shift(breaks)
                array.shift(flipStates)
                array.shift(flipAt)
                array.shift(uidArr)
                array.shift(roleVers)
                array.shift(gradeSince)
            array.push(tops,top)
            array.push(bottoms,bot)
            array.push(sides,side)
            array.push(borns,bar_index)
            array.push(sources,src)
            array.push(strength,1)
            array.push(breaks,0)
            array.push(flipStates,0)
            array.push(flipAt,na)
            // At most one insertion for each source and side per candle.
            array.push(uidArr,bar_index*64 + src*2 + (side == 1 ? 1 : 0))
            array.push(roleVers,0)
            array.push(gradeSince,bar_index)
            inserted := true
    inserted

float upperSwingSlope = not na(previousSwingHigh) and highAt > prevHighAt ? (swingHigh - previousSwingHigh) / (highAt - prevHighAt) : na
float lowerSwingSlope = not na(previousSwingLow) and lowAt > prevLowAt ? (swingLow - previousSwingLow) / (lowAt - prevLowAt) : na
float projectedUpperTrend = not na(upperSwingSlope) ? swingHigh + (bar_index - highAt) * upperSwingSlope : na
float projectedLowerTrend = not na(lowerSwingSlope) ? swingLow + (bar_index - lowAt) * lowerSwingSlope : na
bool swingReady = not na(swingHigh) and not na(swingLow) and swingHigh > swingLow
bool swingUp = swingReady and highAt > lowAt
bool swingDown = swingReady and lowAt > highAt
float fibL50 = swingUp ? swingHigh - 0.5 * (swingHigh - swingLow) : na
float fibL618 = swingUp ? swingHigh - 0.618 * (swingHigh - swingLow) : na
float fibS50 = swingDown ? swingLow + 0.5 * (swingHigh - swingLow) : na
float fibS618 = swingDown ? swingLow + 0.618 * (swingHigh - swingLow) : na

bool upBreak = barstate.isconfirmed and not na(swingHigh) and close > swingHigh and close[1] <= swingHigh
bool dnBreak = barstate.isconfirmed and not na(swingLow) and close < swingLow and close[1] >= swingLow
var int structureDir = 0
bool upCHOCH = upBreak and structureDir == -1
bool dnCHOCH = dnBreak and structureDir == 1
if barstate.isconfirmed
    if upBreak
        structureDir := 1
    if dnBreak
        structureDir := -1
int agoUp = ta.barssince(upBreak)
int agoDn = ta.barssince(dnBreak)
int agoUpCh = ta.barssince(upCHOCH)
int agoDnCh = ta.barssince(dnCHOCH)
bool upFresh = not na(agoUp) and agoUp <= 8
bool dnFresh = not na(agoDn) and agoDn <= 8
bool upChFresh = not na(agoUpCh) and agoUpCh <= 8
bool dnChFresh = not na(agoDnCh) and agoDnCh <= 8

float span = math.max(high - low, syminfo.mintick)
float body = math.abs(close - open)
float upperWick = high - math.max(close, open)
float lowerWick = math.min(close, open) - low
bool bullSweep = not na(swingLow) and low < swingLow and close > swingLow and close > open
bool bearSweep = not na(swingHigh) and high > swingHigh and close < swingHigh and close < open
bool bullEngulf = close > open and close[1] < open[1] and open <= close[1] and close >= open[1]
bool bearEngulf = close < open and close[1] > open[1] and open >= close[1] and close <= open[1]
bool bullReject = close > open and lowerWick >= span * 0.38 and lowerWick > body * 1.2 and close >= low + span * 0.6
bool bearReject = close < open and upperWick >= span * 0.38 and upperWick > body * 1.2 and close <= high - span * 0.6
bool bullImpulse = close > high[1] and close > open and vRatio >= 1.15
bool bearImpulse = close < low[1] and close < open and vRatio >= 1.15
bool longTrigger = bullEngulf or bullReject or bullImpulse or bullSweep
bool shortTrigger = bearEngulf or bearReject or bearImpulse or bearSweep

if barstate.isconfirmed and goodBars
    // Locked state machine: 0=ACTIVE, 1=FLIPPED_AWAIT_RETEST,
    // 2=INVALIDATED. A flipped zone cannot flip again before confirmation.
    if array.size(zTop) > 0
        for i = 0 to array.size(zTop) - 1
            int dir = array.get(zSide,i)
            float top = array.get(zTop,i)
            float bot = array.get(zBot,i)
            int state = array.get(zFlip,i)
            if state == 0
                bool crossed = dir == 1 ? close < bot-breakATR*atr : close > top+breakATR*atr
                int sequence = crossed ? array.get(zBreaks,i)+1 : 0
                if sequence >= 2
                    array.set(zSide,i,-dir)
                    array.set(zRoleVersion,i,array.get(zRoleVersion,i)+1)
                    array.set(zFlip,i,1)
                    array.set(zFlipAt,i,bar_index)
                    array.set(zGradeSince,i,bar_index)
                    array.set(zSource,i,1) // old-role evidence must not justify new role
                    array.set(zPower,i,1)
                    sequence := 0
                array.set(zBreaks,i,sequence)
            else if state == 1
                int elapsed = bar_index-nz(array.get(zFlipAt,i),bar_index)
                bool revisit = high >= bot-atr*contactATR and low <= top+atr*contactATR
                bool held = dir == 1 ? close > top : close < bot
                bool failed = dir == 1 ? close < bot-breakATR*atr : close > top+breakATR*atr
                if elapsed > 0 and revisit and held
                    array.set(zFlip,i,0)
                    array.set(zGradeSince,i,bar_index)
                else if failed or elapsed > retestExpiry
                    array.set(zFlip,i,2)
    // Confirmed swing: price values refer to pivot bar, but zone birth = NOW.
    float zoneW = math.max(atr * zoneHalfATR, syminfo.mintick * 3)
    if not na(newPL)
        f_putZone(zTop, zBot, zSide, zBirth, zSource, zPower, zBreaks, zFlip, zFlipAt, zUID, zRoleVersion, zGradeSince, newPL + zoneW, newPL - zoneW * 0.15, 1, 2, atr * mergeATR, maxZones)
    if not na(newPH)
        f_putZone(zTop, zBot, zSide, zBirth, zSource, zPower, zBreaks, zFlip, zFlipAt, zUID, zRoleVersion, zGradeSince, newPH + zoneW * 0.15, newPH - zoneW, -1, 2, atr * mergeATR, maxZones)
    // 3-candle imbalances: low > high[2] is bullish FVG, inverse bearish.
    if low > high[2] and low - high[2] >= atr * gapMinATR
        f_putZone(zTop, zBot, zSide, zBirth, zSource, zPower, zBreaks, zFlip, zFlipAt, zUID, zRoleVersion, zGradeSince, low, high[2], 1, 3, atr * mergeATR, maxZones)
    if high < low[2] and low[2] - high >= atr * gapMinATR
        f_putZone(zTop, zBot, zSide, zBirth, zSource, zPower, zBreaks, zFlip, zFlipAt, zUID, zRoleVersion, zGradeSince, low[2], high, -1, 3, atr * mergeATR, maxZones)
    // Simple OB: last opposite candle before confirmed BOS / CHOCH.
    // This is a rule-based proxy, not institution order-book data.
    if upBreak
        for j = 1 to 5
            if close[j] < open[j] and high[j] - low[j] < atr * 2.5
                f_putZone(zTop, zBot, zSide, zBirth, zSource, zPower, zBreaks, zFlip, zFlipAt, zUID, zRoleVersion, zGradeSince, open[j], low[j], 1, 5, atr * mergeATR, maxZones)
                break
    if dnBreak
        for j = 1 to 5
            if close[j] > open[j] and high[j] - low[j] < atr * 2.5
                f_putZone(zTop, zBot, zSide, zBirth, zSource, zPower, zBreaks, zFlip, zFlipAt, zUID, zRoleVersion, zGradeSince, high[j], open[j], -1, 5, atr * mergeATR, maxZones)
                break
    // Expired zones are removed from memory. Reverse deletion avoids skips.
    if array.size(zTop) > 0
        for i = array.size(zTop) - 1 to 0
            if bar_index - array.get(zBirth, i) > zoneMaxAge
                array.remove(zTop, i)
                array.remove(zBot, i)
                array.remove(zSide, i)
                array.remove(zBirth, i)
                array.remove(zPower, i)
                array.remove(zSource, i)
                array.remove(zBreaks, i)
                array.remove(zFlip, i)
                array.remove(zFlipAt, i)
                array.remove(zUID, i)
                array.remove(zRoleVersion, i)
                array.remove(zGradeSince, i)

// 05B. CONFLUENCE EVIDENCE LEDGER (per permanent Zone UID + role version)
// --------------------------------------------------------------------------
// Families are BIT FLAGS: structure(1), Fib/Gann(2), regression/trend(4),
// volume/volume profile(8), prior D/W levels(16), pattern(32), candle hold(64).
// Each family counts ONCE. 1-min price twins from multiple indicators do NOT
// create extra independent votes. Evidence is recorded on confirmed candles.
// Opposite volume and confirmed broken closes can remove evidence; role flips
// create a NEW ledger key, preventing prior support votes from endorsing shorts.
f_confluenceHas(int mask, int flag) =>
    int(math.floor(mask / flag)) % 2 == 1
f_confluenceCount(int mask) =>
    (f_confluenceHas(mask,1)?1:0) + (f_confluenceHas(mask,2)?1:0) + (f_confluenceHas(mask,4)?1:0) + (f_confluenceHas(mask,8)?1:0) + (f_confluenceHas(mask,16)?1:0) + (f_confluenceHas(mask,32)?1:0) + (f_confluenceHas(mask,64)?1:0)
f_confluenceTier(int mask) =>
    int cnt = f_confluenceCount(mask)
    bool highGate = f_confluenceHas(mask,8) and f_confluenceHas(mask,16) and f_confluenceHas(mask,64)
    cnt >= 6 and highGate ? 5 : cnt >= 4 ? 4 : cnt >= 3 ? 3 : cnt >= 2 ? 2 : 1
f_confluenceName(int rank) =>
    rank == 5 ? "최상위 합류" : rank == 4 ? "강한 합류" : rank == 3 ? "다중 합류" : rank == 2 ? "초기 합류" : "기초 근거"
f_confluenceExplain(int mask) =>
    string t = ""
    t += f_confluenceHas(mask,1) ? "구조 " : ""
    t += f_confluenceHas(mask,2) ? "피보/갠 " : ""
    t += f_confluenceHas(mask,4) ? "추세 " : ""
    t += f_confluenceHas(mask,8) ? "VOL/VP " : ""
    t += f_confluenceHas(mask,16) ? "일·주봉 " : ""
    t += f_confluenceHas(mask,32) ? "패턴 " : ""
    t += f_confluenceHas(mask,64) ? "안착 " : ""
    t == "" ? "합류 근거 부족" : t

            color evtCol = newTier==0 ? color.gray : newTier>oldTier ? color.rgb(218,173,65) : color.rgb(226,88,88)

// Rank every confirmed active zone at its fixed midpoint; append previous
// day/week levels, confirmed swings, regression/Fibonacci extensions.
// Filter to valid positive prices in ATR range; remove crowded duplicates.
f_pickRoute(array<float> candidates, bool up, float reference, float spacing, float maxDistance) =>
    float first = na
    float secondBest = na
    float third = na
    float last = na
    for stepIndex = 0 to 2
        float best = na
        if array.size(candidates) > 0
            for candidateIndex = 0 to array.size(candidates) - 1
                float p = array.get(candidates,candidateIndex)
                bool valid = not na(p) and p > 0 and math.abs(p-reference) <= maxDistance
                bool inDirection = up ? p > reference + spacing : p < reference - spacing
                bool newStep = na(last) or (up ? p > last + spacing : p < last - spacing)
                bool improves = na(best) or (up ? p < best : p > best)
                if valid and inDirection and newStep and improves
                    best := p
        if stepIndex == 0
            first := best
        else if stepIndex == 1
            secondBest := best
        else
            third := best
        if na(best)
            break
        last := best
    [first, secondBest, third]
